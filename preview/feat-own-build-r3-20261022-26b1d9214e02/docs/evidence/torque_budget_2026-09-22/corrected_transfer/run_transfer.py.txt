def check(item):
 name,args=item;cmd=[GODOT,'--headless','--path',str(ROOT),'--fixed-fps','60','--script',f'tests/{name}.gd']+(['--']+args if args else [])
 r=subprocess.run(cmd,capture_output=True,text=True,timeout=180);label=name+('-'+'-'.join(args) if args else '')
 (out/(label+'.log')).write_text(r.stdout+r.stderr)

 print(label,r.returncode,flush=True);return entry
