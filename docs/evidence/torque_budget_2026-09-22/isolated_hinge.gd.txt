extends SceneTree


func _initialize() -> void:
	_run.call_deferred()


func _run() -> void:
	for divisor: float in [1.0, 64.0]:
		var viewport := SubViewport.new()
		viewport.own_world_3d = true
		root.add_child(viewport)
		var anchor := StaticBody3D.new()
		viewport.add_child(anchor)
		var body := RigidBody3D.new()
		body.mass = 1.0
		body.inertia = Vector3.ONE
		body.gravity_scale = 0.0
		body.angular_damp_mode = RigidBody3D.DAMP_MODE_REPLACE
		body.angular_damp = 0.0
		body.linear_damp_mode = RigidBody3D.DAMP_MODE_REPLACE
		body.linear_damp = 0.0
		body.can_sleep = false
		viewport.add_child(body)
		var shape := CollisionShape3D.new()
		shape.shape = SphereShape3D.new()
		body.add_child(shape)
		var joint := HingeJoint3D.new()
		viewport.add_child(joint)
		joint.node_a = joint.get_path_to(anchor)
		joint.node_b = joint.get_path_to(body)
		joint.set_flag(HingeJoint3D.FLAG_USE_LIMIT, false)
		joint.set_param(HingeJoint3D.PARAM_MOTOR_TARGET_VELOCITY, 100.0)
		joint.set_param(HingeJoint3D.PARAM_MOTOR_MAX_IMPULSE, 0.25 / 60.0 / divisor)
		for frame: int in 3:
			await physics_frame
		var previous: float = body.angular_velocity.z
		joint.set_flag(HingeJoint3D.FLAG_ENABLE_MOTOR, true)
		for frame: int in 6:
			await physics_frame
			var current: float = body.angular_velocity.z
			print("MOTOR_BUDGET ", JSON.stringify({"divisor":divisor,"frame":frame,"velocity_z":current,
				"effective_torque_nm":absf(current-previous)*60.0,"specified_torque_nm":0.25,
				"solver_iterations":ProjectSettings.get_setting("physics/3d/solver/solver_iterations")}))
			previous = current
		viewport.free()
		await physics_frame
	quit()
